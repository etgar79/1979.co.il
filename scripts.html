דף הבית סקריפטים

בדיקת מידע מחשב

מערכת
▼
מציג מידע מפורט על המחשב - שם, מערכת הפעלה, מעבד, זיכרון ודיסקים.
# בדיקת מידע מחשב
$computerInfo = Get-ComputerInfo
Write-Host "Computer Name: $($computerInfo.CsName)" -ForegroundColor Cyan
Write-Host "OS: $($computerInfo.OsName)"
Write-Host "RAM: $([math]::Round($computerInfo.CsTotalPhysicalMemory / 1GB, 2)) GB"

# דיסקים
Get-Disk | Format-Table Number, FriendlyName, @{N='Size(GB)';E={[math]::Round($_.Size / 1GB, 2)}}

בדיקת חיבורי רשת

רשת
▼
בודק חיבורים פעילים, כתובות IP ומבצע בדיקת DNS.
# בדיקת חיבורי רשת
Get-NetIPAddress -AddressFamily IPv4 |
  Where-Object { $_.IPAddress -ne "127.0.0.1" } |
  Format-Table InterfaceAlias, IPAddress, PrefixLength

# בדיקת DNS
Resolve-DnsName "google.com" | Format-Table Name, Type, IPAddress

# Ping מהיר
Test-Connection -ComputerName "8.8.8.8" -Count 4

חיבור ל-Microsoft 365

אופיס 365
▼
מתחבר לשירותי Microsoft 365 (Exchange Online, Azure AD) ומציג מידע בסיסי.
# התקנת מודולים (פעם ראשונה בלבד)
Install-Module -Name ExchangeOnlineManagement -Force
Install-Module -Name AzureAD -Force

# התחברות
Connect-ExchangeOnline
Connect-AzureAD

# רשימת משתמשים
Get-AzureADUser -All $true |
  Select-Object DisplayName, UserPrincipalName, AccountEnabled |
  Format-Table -AutoSize
